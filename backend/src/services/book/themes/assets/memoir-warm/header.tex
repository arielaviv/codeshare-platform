%% Memoir Warm — Crimson Pro throughout. Right-italic-smallcaps opener,
%% 2-line drop caps, spaced-dot scene break. Vintage, intimate.

\newfontfamily\chapterfont{Crimson Pro}

%% Memoir accent — burnt sienna.
\definecolor{memoirAccent}{HTML}{7A3B2E}

%% Chapter opener — right-aligned italic small-caps.
\titleformat{\chapter}[display]
  {\normalfont\chapterfont\filleft}
  {\normalfont\chapterfont\itshape\scshape\large Chapter \thechapter}
  {0.8em}
  {\chapterfont\Huge\itshape}
\titlespacing*{\chapter}{0pt}{42pt}{32pt}

%% Drop cap — 2-line, accent color, italic for memoir warmth.
\renewcommand{\DefaultLoversize}{0.14}
\renewcommand{\DefaultLraise}{0.02}
\setcounter{DefaultLines}{2}
\renewcommand{\LettrineFontHook}{\color{memoirAccent}\chapterfont\itshape}

%% Scene break — three spaced middle dots.
\renewcommand{\sceneBreakSymbol}{\chapterfont\itshape\large\textperiodcentered\kern1em\textperiodcentered\kern1em\textperiodcentered}

%% Running heads — title verso, chapter recto. Centered folios.
\makeatletter
\pagestyle{fancy}
\fancyhf{}
\fancyhead[LE]{\small\chapterfont\itshape\@title}
\fancyhead[RO]{\small\chapterfont\itshape\nouppercase{\leftmark}}
\fancyfoot[C]{\chapterfont\thepage}
\renewcommand{\headrulewidth}{0pt}
\renewcommand{\chaptermark}[1]{\markboth{#1}{}}
\makeatother

\fancypagestyle{plain}{%
  \fancyhf{}%
  \fancyfoot[C]{\chapterfont\thepage}%
  \renewcommand{\headrulewidth}{0pt}%
}
